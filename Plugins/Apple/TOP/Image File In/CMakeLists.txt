cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS deployment target")
set(CMAKE_OSX_ARCHITECTURES "arm64")
project(AppleImageFileIn LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "Apple Image File In requires macOS")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/plugin" CACHE PATH "Plugin bundle output directory")

find_library(IMAGEIO_FRAMEWORK ImageIO REQUIRED)
find_library(COREIMAGE_FRAMEWORK CoreImage REQUIRED)
find_library(COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
find_library(AVFOUNDATION_FRAMEWORK AVFoundation REQUIRED)
find_library(COREVIDEO_FRAMEWORK CoreVideo REQUIRED)
find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)

set(TD_APPLICATION "/Applications/TouchDesigner.app" CACHE PATH "TouchDesigner application to build against")
set(TD_PYTHON_ROOT "${TD_APPLICATION}/Contents/Frameworks/Python.framework")
file(GLOB TD_PYTHON_HEADERS "${TD_PYTHON_ROOT}/Versions/Current/include/python*/Python.h")
list(LENGTH TD_PYTHON_HEADERS TD_PYTHON_HEADER_COUNT)
if(NOT TD_PYTHON_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one Python header under ${TD_PYTHON_ROOT}/Versions/Current/include; set TD_APPLICATION to your TouchDesigner app")
endif()
list(GET TD_PYTHON_HEADERS 0 TD_PYTHON_HEADER)
get_filename_component(TD_PYTHON_INCLUDE "${TD_PYTHON_HEADER}" DIRECTORY)
unset(TD_PYTHON_FRAMEWORK CACHE)
find_library(TD_PYTHON_FRAMEWORK Python PATHS "${TD_APPLICATION}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Host.py")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Host.py" IMAGE_HOST_SCRIPT)
configure_file(HostScript.h.in HostScript.h @ONLY)

add_library(ImageReader STATIC ImageData.cpp Decoder.mm)
set_target_properties(ImageReader PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ImageReader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(ImageReader PRIVATE -Wall -Wextra "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(ImageReader PUBLIC "${IMAGEIO_FRAMEWORK}" "${COREIMAGE_FRAMEWORK}"
    "${COREGRAPHICS_FRAMEWORK}" "${AVFOUNDATION_FRAMEWORK}" "${COREVIDEO_FRAMEWORK}"
    "${FOUNDATION_FRAMEWORK}" "${COREFOUNDATION_FRAMEWORK}")
add_library(AppleImageFileIn MODULE AppleImageFileIn.cpp Host.cpp)
target_link_libraries(AppleImageFileIn PRIVATE ImageReader "${TD_PYTHON_FRAMEWORK}")
target_include_directories(AppleImageFileIn PRIVATE "${TD_PYTHON_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(AppleImageFileIn PROPERTIES PREFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
option(TD_ADHOC_SIGN "Ad-hoc sign the local bundle" ON)
if(TD_ADHOC_SIGN)
    find_program(CODESIGN_EXECUTABLE codesign REQUIRED)
    add_custom_command(TARGET AppleImageFileIn POST_BUILD
        COMMAND "${CODESIGN_EXECUTABLE}" --force --sign - "$<TARGET_BUNDLE_DIR:AppleImageFileIn>"
        COMMENT "Ad-hoc signing AppleImageFileIn.plugin" VERBATIM)
endif()
option(TD_BUILD_TESTS "Build decoding, engine, host and docking tests" OFF)
if(TD_BUILD_TESTS)
    enable_testing()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(ImageReaderTests tests/reader.mm)
    target_compile_options(ImageReaderTests PRIVATE -fobjc-arc)
    target_link_libraries(ImageReaderTests PRIVATE ImageReader)
    add_test(NAME ImageReader COMMAND ImageReaderTests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(ImageReader PROPERTIES TIMEOUT 120)
    add_executable(ImageEngineTests tests/engine.cpp)
    target_link_libraries(ImageEngineTests PRIVATE ImageReader)
    add_test(NAME ImageEngine COMMAND ImageEngineTests "${CMAKE_CURRENT_BINARY_DIR}/engine-fixtures")
    add_executable(ImageHostTests tests/host.cpp AppleImageFileIn.cpp)
    target_link_libraries(ImageHostTests PRIVATE ImageReader)
    add_test(NAME ImageHost COMMAND ImageHostTests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(ImageHost PROPERTIES DEPENDS ImageReader)
    add_test(NAME ImageDocking COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/docking.py")
endif()
