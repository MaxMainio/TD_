cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
set(CMAKE_OSX_ARCHITECTURES "arm64")
project(AppleFaceDetection LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "Apple Face Detection requires macOS and Apple Vision")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/plugin" CACHE PATH "Plugin bundle output directory")

find_library(VISION_FRAMEWORK Vision REQUIRED)
find_library(COREVIDEO_FRAMEWORK CoreVideo REQUIRED)
find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)

set(TD_APPLICATION "/Applications/TouchDesigner.app" CACHE PATH "TouchDesigner application to build against")
set(TD_PYTHON_ROOT "${TD_APPLICATION}/Contents/Frameworks/Python.framework")
file(GLOB TD_PYTHON_HEADERS "${TD_PYTHON_ROOT}/Versions/Current/include/python*/Python.h")
list(LENGTH TD_PYTHON_HEADERS TD_PYTHON_HEADER_COUNT)
if(NOT TD_PYTHON_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one Python header under ${TD_PYTHON_ROOT}/Versions/Current/include; set TD_APPLICATION to your TouchDesigner app")
endif()
list(GET TD_PYTHON_HEADERS 0 TD_PYTHON_HEADER)
get_filename_component(TD_PYTHON_INCLUDE "${TD_PYTHON_HEADER}" DIRECTORY)
unset(TD_PYTHON_FRAMEWORK CACHE)
find_library(TD_PYTHON_FRAMEWORK Python PATHS "${TD_APPLICATION}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py" FACE_DOCKING_SCRIPT)
configure_file(DockingScript.h.in DockingScript.h @ONLY)

add_library(FaceData STATIC FaceData.cpp)
set_target_properties(FaceData PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(FaceData PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(FaceData PRIVATE -Wall -Wextra -Wpedantic)
add_library(FaceVision STATIC VisionDetector.mm)
set_target_properties(FaceVision PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(FaceVision PRIVATE -fobjc-arc -Wall -Wextra)
target_link_libraries(FaceVision PUBLIC FaceData "${VISION_FRAMEWORK}" "${COREVIDEO_FRAMEWORK}" "${FOUNDATION_FRAMEWORK}" "${COREFOUNDATION_FRAMEWORK}")

add_library(AppleFaceDetection MODULE AppleFaceDetection.mm Docking.cpp)
target_link_libraries(AppleFaceDetection PRIVATE FaceVision "${TD_PYTHON_FRAMEWORK}")
target_include_directories(AppleFaceDetection PRIVATE "${TD_PYTHON_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(AppleFaceDetection PROPERTIES PREFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
option(TD_ADHOC_SIGN "Ad-hoc sign the local bundle" ON)
if(TD_ADHOC_SIGN)
    find_program(CODESIGN_EXECUTABLE codesign REQUIRED)
    add_custom_command(TARGET AppleFaceDetection POST_BUILD
        COMMAND "${CODESIGN_EXECUTABLE}" --force --sign - "$<TARGET_BUNDLE_DIR:AppleFaceDetection>"
        COMMENT "Ad-hoc signing AppleFaceDetection.plugin" VERBATIM)
endif()

option(TD_BUILD_TESTS "Build mask, landmarks, host, docking and Vision tests" OFF)
if(TD_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    enable_testing()
    add_executable(FaceDataTests tests/data.cpp)
    target_link_libraries(FaceDataTests PRIVATE FaceData)
    add_test(NAME FaceMasksAndLandmarks COMMAND FaceDataTests)
    add_executable(FaceHostTests tests/host.cpp AppleFaceDetection.mm)
    target_link_libraries(FaceHostTests PRIVATE FaceData)
    add_test(NAME FaceHostLifecycle COMMAND FaceHostTests)
    add_test(NAME FaceDockingHelper COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/docking.py")
    add_executable(FaceVisionTests tests/vision.mm)
    target_link_libraries(FaceVisionTests PRIVATE FaceVision)
    add_test(NAME FaceVisionSmoke COMMAND FaceVisionTests)
    set_tests_properties(FaceVisionSmoke PROPERTIES TIMEOUT 120)
endif()
