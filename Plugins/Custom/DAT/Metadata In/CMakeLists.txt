cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
project(MetaData_In VERSION 0.1.1 LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "MetaData_In currently supports macOS only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/plugin" CACHE PATH "Bundle output directory")
set(TD_APPLICATION "/Applications/TouchDesigner.app" CACHE PATH "TouchDesigner installation")
set(METADATA_READER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/readers" CACHE PATH "Prepared bundled reader resources")
find_package(Threads REQUIRED)
find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND /usr/bin/perl -MCompress::Zlib -MDigest::MD5 -MDigest::SHA -e "exit 0" RESULT_VARIABLE PERL_CHECK)
if(NOT PERL_CHECK EQUAL 0)
    message(FATAL_ERROR "System Perl with Compress::Zlib, Digest::MD5 and Digest::SHA is required")
endif()
foreach(READER ffprobe exiftool/exiftool versions.json dependencies.json)
    if(NOT EXISTS "${METADATA_READER_ROOT}/${READER}")
        message(FATAL_ERROR "Missing bundled reader ${READER}. Run python3 scripts/prepare-dependencies.py from this plugin folder, or set METADATA_READER_ROOT.")
    endif()
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" EXPECTED_LOCK_HASH)
file(SHA256 "${METADATA_READER_ROOT}/dependencies.json" PREPARED_LOCK_HASH)
if(NOT EXPECTED_LOCK_HASH STREQUAL PREPARED_LOCK_HASH)
    message(FATAL_ERROR "Prepared readers do not match dependencies.json; prepare them in a fresh build root")
endif()
execute_process(COMMAND /usr/bin/lipo -archs "${METADATA_READER_ROOT}/ffprobe"
    OUTPUT_VARIABLE READER_ARCHS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(REQUIRED_ARCHS ${CMAKE_OSX_ARCHITECTURES})
if(NOT REQUIRED_ARCHS)
    set(REQUIRED_ARCHS ${CMAKE_SYSTEM_PROCESSOR})
endif()
foreach(ARCH IN LISTS REQUIRED_ARCHS)
    if(NOT " ${READER_ARCHS} " MATCHES " ${ARCH} ")
        message(FATAL_ERROR "Bundled ffprobe lacks ${ARCH}; rerun prepare-dependencies.py --architectures ${REQUIRED_ARCHS}")
    endif()
endforeach()
set(TD_PYTHON_ROOT "${TD_APPLICATION}/Contents/Frameworks/Python.framework")
file(GLOB TD_HEADERS "${TD_PYTHON_ROOT}/Versions/Current/include/python*/Python.h")
list(LENGTH TD_HEADERS TD_HEADER_COUNT)
if(NOT TD_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR "Set TD_APPLICATION to a TouchDesigner installation with Python development headers")
endif()
list(GET TD_HEADERS 0 TD_HEADER)
get_filename_component(TD_PYTHON_INCLUDE "${TD_HEADER}" DIRECTORY)
find_library(TD_PYTHON_FRAMEWORK Python PATHS "${TD_APPLICATION}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Resolver.py")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Resolver.py" METADATA_RESOLVER_SCRIPT)
configure_file(ResolverScript.h.in ResolverScript.h @ONLY)
add_library(MetadataEngine STATIC Metadata.cpp Process.cpp Geometry.cpp Table.cpp)
set_target_properties(MetadataEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(MetadataEngine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(MetadataEngine PUBLIC Threads::Threads)
target_compile_options(MetadataEngine PRIVATE -Wall -Wextra -Wpedantic)
add_library(MetaData_In MODULE MetaDataInDAT.cpp)
target_include_directories(MetaData_In PRIVATE "${TD_PYTHON_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(MetaData_In PRIVATE MetadataEngine "${TD_PYTHON_FRAMEWORK}")
target_compile_options(MetaData_In PRIVATE -Wno-invalid-offsetof)
set_target_properties(MetaData_In PROPERTIES PREFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
add_custom_command(TARGET MetaData_In POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${METADATA_READER_ROOT}" "$<TARGET_BUNDLE_DIR:MetaData_In>/Contents/Resources"
    COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_LICENSES.txt" "$<TARGET_BUNDLE_DIR:MetaData_In>/Contents/Resources/licenses/THIRD_PARTY_LICENSES.txt"
    COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_SOURCE_DIR}/vendor/LICENSE-json.txt" "$<TARGET_BUNDLE_DIR:MetaData_In>/Contents/Resources/licenses/LICENSE-json.txt"
    VERBATIM)
option(TD_ADHOC_SIGN "Ad-hoc sign the local plugin" ON)
if(TD_ADHOC_SIGN)
    add_custom_command(TARGET MetaData_In POST_BUILD
        COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:MetaData_In>/Contents/Resources/ffprobe"
        COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:MetaData_In>" VERBATIM)
endif()
option(TD_BUILD_TESTS "Build metadata reader and lifecycle tests" OFF)
if(TD_BUILD_TESTS)
    enable_testing()
    add_executable(MetadataTests tests/tests.cpp)
    target_link_libraries(MetadataTests PRIVATE MetadataEngine)
    add_executable(MetadataProbe tests/probe.cpp)
    target_link_libraries(MetadataProbe PRIVATE MetadataEngine)
    add_test(NAME MetadataCore COMMAND MetadataTests "${CMAKE_CURRENT_BINARY_DIR}/test-data" "${METADATA_READER_ROOT}")
    add_test(NAME MetadataResolver COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/resolver.py")
    add_test(NAME MetadataMedia COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/media.py"
        "$<TARGET_FILE:MetadataProbe>" "${METADATA_READER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/media-data")
    set_tests_properties(MetadataCore MetadataResolver MetadataMedia PROPERTIES TIMEOUT 180)
endif()
