cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(ConvexHull LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/plugin" CACHE PATH "Plugin bundle output directory")

add_library(HullEngine STATIC HullEngine.cpp HullMeasurements.cpp HullInfoDAT.cpp)
set_target_properties(HullEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(HullEngine PUBLIC Threads::Threads)
target_include_directories(HullEngine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(HullEngine PRIVATE -ffp-contract=off -fno-fast-math -Wall -Wextra -Wpedantic)
endif()

if(NOT APPLE)
    message(FATAL_ERROR "Convex Hull currently supports macOS with TouchDesigner installed")
endif()
set(TD_APPLICATION "/Applications/TouchDesigner.app" CACHE PATH "TouchDesigner application to build against")
set(TD_PYTHON_ROOT "${TD_APPLICATION}/Contents/Frameworks/Python.framework")
file(GLOB TD_PYTHON_HEADERS "${TD_PYTHON_ROOT}/Versions/Current/include/python*/Python.h")
list(LENGTH TD_PYTHON_HEADERS TD_PYTHON_HEADER_COUNT)
if(NOT TD_PYTHON_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one Python header under ${TD_PYTHON_ROOT}/Versions/Current/include; set TD_APPLICATION to your TouchDesigner app")
endif()
list(GET TD_PYTHON_HEADERS 0 TD_PYTHON_HEADER)
get_filename_component(TD_PYTHON_INCLUDE "${TD_PYTHON_HEADER}" DIRECTORY)
unset(TD_PYTHON_FRAMEWORK CACHE)
find_library(TD_PYTHON_FRAMEWORK Python PATHS "${TD_APPLICATION}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py" HULL_DOCKING_SCRIPT)
configure_file(DockingScript.h.in DockingScript.h @ONLY)

add_library(ConvexHull MODULE ConvexHullTOP.cpp Parameters.cpp Docking.cpp)
target_link_libraries(ConvexHull PRIVATE HullEngine)
target_link_libraries(ConvexHull PRIVATE "${TD_PYTHON_FRAMEWORK}")
target_include_directories(ConvexHull PRIVATE "${TD_PYTHON_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(ConvexHull PROPERTIES PREFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
if(APPLE)
    option(TD_ADHOC_SIGN "Ad-hoc sign the local bundle" ON)
    if(TD_ADHOC_SIGN)
        find_program(CODESIGN_EXECUTABLE codesign REQUIRED)
        add_custom_command(TARGET ConvexHull POST_BUILD
            COMMAND "${CODESIGN_EXECUTABLE}" --force --sign - "$<TARGET_BUNDLE_DIR:ConvexHull>"
            COMMENT "Ad-hoc signing ConvexHull.plugin" VERBATIM)
    endif()
endif()
option(TD_BUILD_TESTS "Build reference, packing, failure and lifecycle tests" OFF)
if(TD_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    enable_testing()
    add_executable(HullTests tests/reference.cpp)
    target_link_libraries(HullTests PRIVATE HullEngine)
    add_test(NAME HullReferenceAndLifecycle COMMAND HullTests)
    set_tests_properties(HullReferenceAndLifecycle PROPERTIES TIMEOUT 120)
    add_executable(HullFailureTests tests/failures.cpp)
    target_link_libraries(HullFailureTests PRIVATE HullEngine)
    add_test(NAME HullAllocationFailures COMMAND HullFailureTests)
    set_tests_properties(HullAllocationFailures PROPERTIES TIMEOUT 120)
    add_executable(HullBenchmark tests/benchmark.cpp)
    target_link_libraries(HullBenchmark PRIVATE HullEngine)
    add_executable(HullMeasurementTests tests/measurements.cpp)
    target_link_libraries(HullMeasurementTests PRIVATE HullEngine)
    add_test(NAME HullMeasurements COMMAND HullMeasurementTests)
    set_tests_properties(HullMeasurements PROPERTIES TIMEOUT 120)
    add_test(NAME HullDockingHelper COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/docking.py")
endif()
