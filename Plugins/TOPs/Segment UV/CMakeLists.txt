cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE)
    # Current Homebrew OpenCV used for validation also requires macOS 15.
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(SegmentUV LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
set(OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/plugin" CACHE PATH "Plugin bundle output directory")

add_library(SegmentEngine STATIC SegmentEngine.cpp SegmentInfoDAT.cpp)
set_target_properties(SegmentEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(SegmentEngine PUBLIC ${OpenCV_LIBS})
target_include_directories(SegmentEngine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" ${OpenCV_INCLUDE_DIRS})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(SegmentEngine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NOT APPLE)
    message(FATAL_ERROR "Segment UV currently supports macOS with TouchDesigner installed")
endif()
set(TD_APPLICATION "/Applications/TouchDesigner.app" CACHE PATH "TouchDesigner application to build against")
set(TD_PYTHON_ROOT "${TD_APPLICATION}/Contents/Frameworks/Python.framework")
file(GLOB TD_PYTHON_HEADERS "${TD_PYTHON_ROOT}/Versions/Current/include/python*/Python.h")
list(LENGTH TD_PYTHON_HEADERS TD_PYTHON_HEADER_COUNT)
if(NOT TD_PYTHON_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one Python header under ${TD_PYTHON_ROOT}/Versions/Current/include; set TD_APPLICATION to your TouchDesigner app")
endif()
list(GET TD_PYTHON_HEADERS 0 TD_PYTHON_HEADER)
get_filename_component(TD_PYTHON_INCLUDE "${TD_PYTHON_HEADER}" DIRECTORY)
unset(TD_PYTHON_FRAMEWORK CACHE)
find_library(TD_PYTHON_FRAMEWORK Python PATHS "${TD_APPLICATION}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Docking.py" SEGMENT_DOCKING_SCRIPT)
configure_file(DockingScript.h.in DockingScript.h @ONLY)

add_library(SegmentUV MODULE SegmentUV.cpp Parameters.cpp Docking.cpp)
target_link_libraries(SegmentUV PRIVATE SegmentEngine)
target_link_libraries(SegmentUV PRIVATE "${TD_PYTHON_FRAMEWORK}")
target_include_directories(SegmentUV PRIVATE "${TD_PYTHON_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(SegmentUV PROPERTIES PREFIX "" BUNDLE TRUE BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(SegmentUV PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY_${config} "${OUTPUT_DIRECTORY}"
        RUNTIME_OUTPUT_DIRECTORY_${config} "${OUTPUT_DIRECTORY}")
endforeach()
if(APPLE)
    option(TD_ADHOC_SIGN "Ad-hoc sign the local bundle" ON)
    if(TD_ADHOC_SIGN)
        find_program(CODESIGN_EXECUTABLE codesign REQUIRED)
        add_custom_command(TARGET SegmentUV POST_BUILD
            COMMAND "${CODESIGN_EXECUTABLE}" --force --sign - "$<TARGET_BUNDLE_DIR:SegmentUV>"
            COMMENT "Ad-hoc signing SegmentUV.plugin" VERBATIM)
    endif()
endif()
option(TD_BUILD_TESTS "Build image parity, measurements, host simulation and docking tests" OFF)
if(TD_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    enable_testing()
    add_executable(SegmentTests tests/reference.cpp tests/legacy.cpp)
    target_link_libraries(SegmentTests PRIVATE SegmentEngine)
    add_test(NAME SegmentReferenceAndMeasurements COMMAND SegmentTests)
    add_executable(SegmentHostTests tests/host.cpp SegmentUV.cpp Parameters.cpp)
    target_link_libraries(SegmentHostTests PRIVATE SegmentEngine)
    add_test(NAME SegmentHostLifecycle COMMAND SegmentHostTests)
    add_executable(SegmentFailureTests tests/failures.cpp)
    target_link_libraries(SegmentFailureTests PRIVATE SegmentEngine)
    add_test(NAME SegmentAllocationFailures COMMAND SegmentFailureTests)
    add_test(NAME SegmentDockingHelper COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/docking.py")
    set_tests_properties(SegmentReferenceAndMeasurements SegmentHostLifecycle SegmentAllocationFailures SegmentDockingHelper PROPERTIES TIMEOUT 120)
    add_executable(SegmentBenchmark tests/benchmark.cpp tests/legacy.cpp)
    target_link_libraries(SegmentBenchmark PRIVATE SegmentEngine)
endif()
